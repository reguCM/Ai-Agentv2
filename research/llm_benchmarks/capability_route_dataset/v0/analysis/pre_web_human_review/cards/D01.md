### **2. ローカルコードでの実装例（Python）**
```python
import requests
from bs4 import BeautifulSoup

def web_search(query, url="https://example.com"):
    try:
        # HTTPリクエスト送信
        response = requests.get(url, params={"q": query}, timeout=10)
        response.raise_for_status()  # エラーがあれば例外をスロー

        # HTMLコンテンツを解析
        soup = BeautifulSoup(response.text, "html.parser")

        # 検索結果を抽出（例: <div class="result">内のテキスト）
        results = soup.find_all("div", class_="result")
        extracted_data = [result.get_text(strip=True) for result in results]

        return extracted_data

    except requests.exceptions.RequestException as e:
        print(f"ネットワークエラー: {e}")
        return []
    except Exception as e:
        print(f"予期せぬエラー: {e}")
        return []
```
